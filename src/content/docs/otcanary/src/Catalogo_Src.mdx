```cpp
        [](DBResult* res) {
            // Callback com resultados
        }
    )
);
```

### databasetasks.hpp / databasetasks.cpp

| Propriedade | Valor |
|---|---|
| **Classe** | `DatabaseTask` (base), `QueryTask`, etc |
| **Responsabilidade** | Encapsular operações de DB |
| **Virtual** | `execute()`, `onSuccess()`, `onFailure()` |

```cpp
class DatabaseTask {
    virtual void execute(Database* db) abstract;
    virtual void onSuccess() {}
    virtual void onFailure() {}
};

class QueryTask : public DatabaseTask {
    std::string query;
    std::function<void(DBResult*)> callback;
    
    void execute(Database* db) override {
        DBResult* result = db->query(query);
        callback(result);
    }
};
```
